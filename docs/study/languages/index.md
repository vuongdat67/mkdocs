<script>
// Calculate totals
function updateStats() {
  const languages = [
    { lessons: 125, vocab: 2500, hours: 180 },
    { lessons: 45, vocab: 800, hours: 60 },
    { lessons: 60, vocab: 1200, hours: 85 },
    { lessons: 35, vocab: 600, hours: 45 },
    { lessons: 25, vocab: 400, hours: 30 }
  ];

  const totalLessons = languages.reduce((sum, lang) => sum + lang.lessons, 0);
  const totalVocab = languages.reduce((sum, lang) => sum + lang.vocab, 0);
  const totalHours = languages.reduce((sum, lang) => sum + lang.hours, 0);

  document.getElementById('total-lessons').textContent = totalLessons;
  document.getElementById('total-vocab').textContent = totalVocab.toLocaleString();
  document.getElementById('study-hours').textContent = totalHours;
}

// Initialize
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', updateStats);
} else {
  updateStats();
}
</script>
