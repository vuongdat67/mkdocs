## 🔍 Cách giải từng bước

```python
cipher_hex = "0e0b213f26041e480b26217f27342e175d0e070a3c5b103e2526217f27342e175d0e077e263451150104"
cipher_bytes = bytes.fromhex(cipher_hex)

# Step 1: known plaintext
known_plain = b"crypto{"

# Step 2: derive the key
key = []
for i in range(len(known_plain)):
    k = cipher_bytes[i] ^ known_plain[i]
    key.append(k)

# convert key to bytes
key_bytes = bytes(key)
print(f"[+] Key derived: {key_bytes}")

# Step 3: decrypt full ciphertext using repeating key
plaintext = []
key_len = len(key_bytes)

for i in range(len(cipher_bytes)):
    decrypted_byte = cipher_bytes[i] ^ key_bytes[i % key_len]
    plaintext.append(decrypted_byte)

# convert plaintext to string
plaintext_str = bytes(plaintext).decode()
print(f"[+] Flag: {plaintext_str}")


```
