## Solve this Roman emperor's cipher: SODWH YLUXV PRXVH IORZHU

## Enter four-word solution here

  

``` python

def decode_cipher(text):

    shift = 3

    decoded_text = ""

    for char in text:

        if 'A' <= char <= 'Z':  # Chỉ xử lý chữ in hoa

            shifted_char = chr((ord(char) - shift - 65) % 26 + 65)

            decoded_text += shifted_char

        else:

            decoded_text += char

    return decoded_text

# Decode the given cipher text

cipher_text = "SODWH YLUXV PRXVH IORZHU"

decoded_solution = decode_cipher(cipher_text)

print(decoded_solution)
```
